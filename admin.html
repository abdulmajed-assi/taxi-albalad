-- تفعيل ملحقات PostGIS والبحث النصي CREATE EXTENSION IF NOT EXISTS postgis; CREATE EXTENSION IF NOT EXISTS pg_trgm; CREATE EXTENSION IF NOT EXISTS unaccent; -- 1. جدول المحافظات والتقسيم الإداري CREATE TABLE IF NOT EXISTS administrative_regions ( id SERIAL PRIMARY KEY, name_ar VARCHAR(100) NOT NULL, name_en VARCHAR(100) NOT NULL, parent_id INT REFERENCES administrative_regions(id), level VARCHAR(20) NOT NULL, -- 'governorate', 'city', 'district', 'subdistrict' boundary GEOMETRY(MultiPolygon, 4326), created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP ); -- 2. جدول الأماكن والنقاط الجغرافية (Places Core) CREATE TABLE IF NOT EXISTS places ( id BIGSERIAL PRIMARY KEY, osm_id BIGINT UNIQUE, osm_type VARCHAR(10), -- 'node', 'way', 'relation' name_ar VARCHAR(255) NOT NULL, name_en VARCHAR(255), aliases_ar TEXT[], -- أسماء بديلة بالعربية (مثال: ["المواساة", "مستشفى المواساة"]) aliases_en TEXT[], -- أسماء بديلة بالإنجليزية (مثال: ["Mouwasat", "Al Mouwasat"]) category VARCHAR(50) NOT NULL, subcategory VARCHAR(50), governorate VARCHAR(100) NOT NULL, city VARCHAR(100), district VARCHAR(100), neighborhood VARCHAR(100), street VARCHAR(255), address TEXT, phone VARCHAR(50), latitude DOUBLE PRECISION NOT NULL, longitude DOUBLE PRECISION NOT NULL, location GEOMETRY(Point, 4326) GENERATED ALWAYS AS (ST_SetSRID(ST_MakePoint(longitude, latitude), 4326)) STORED, -- الوزن والأولوية priority_level INT DEFAULT 12, -- 1 للمشافي حتى 12 للأماكن العادية search_weight DOUBLE PRECISION DEFAULT 1.0, searchable BOOLEAN DEFAULT TRUE, -- نص البحث المعياري الموحد (Normalized Search Text) normalized_search_text TEXT, metadata JSONB, version INT DEFAULT 1, created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP, updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP ); -- 3. جدول التحقق من البيانات غير المؤكدة (Needs Verification) CREATE TABLE IF NOT EXISTS places_needs_verification ( id BIGSERIAL PRIMARY KEY, raw_name TEXT, suggested_category TEXT, suggested_governorate TEXT, latitude DOUBLE PRECISION, longitude DOUBLE PRECISION, osm_id BIGINT, missing_fields TEXT[], review_status VARCHAR(20) DEFAULT 'PENDING', -- 'PENDING', 'APPROVED', 'REJECTED' reviewer_notes TEXT, created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP ); -- 4. فهارس الأداء العالي (Spatial & Full-Text Indexes) CREATE INDEX IF NOT EXISTS idx_places_spatial ON places USING GIST(location); CREATE INDEX IF NOT EXISTS idx_places_governorate ON places(governorate); CREATE INDEX IF NOT EXISTS idx_places_category_priority ON places(category, priority_level); CREATE INDEX IF NOT EXISTS idx_places_normalized_trgm ON places USING GIN(normalized_search_text gin_trgm_ops);
